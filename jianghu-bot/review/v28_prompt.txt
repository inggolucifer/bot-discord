```js
const mongoose = require('mongoose');
function getPlayerQuery(userId) {
  if (!userId) return { _id: null };
  const isObjId = mongoose.Types.ObjectId.isValid(userId) && String(new mongoose.Types.ObjectId(userId)) === String(userId);
  return isObjId ? { $or: [{ discordId: userId }, { _id: userId }] } : { discordId: String(userId) };
}
function getUserIdFromReq(req) { return (req.user && (req.user.userId || req.user.id || req.user.discordId)) || null; }
async function loadPlayer(Player, req) { const id = getUserIdFromReq(req); return id ? Player.findOne(getPlayerQuery(id)) : null; }
module.exports = { getPlayerQuery, getUserIdFromReq, loadPlayer };
```
